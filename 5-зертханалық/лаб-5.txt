lst = [1, 2, 3, 4, 5]
reverse_lst = lst[::-1]
print(reverse_lst)
def list_sort(lst):
    return sorted(lst, key=abs, reverse=True)

lst = [-1, 3, -7, 2, 5]
sorted_lst = list_sort(lst)
print(sorted_lst)
def change(lst):
    lst[0], lst[-1] = lst[-1], lst[0]
    return lst

lst = [1, 2, 3, 4, 5]
changed_lst = change(lst)
print(changed_lst)
